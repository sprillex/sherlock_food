Android Build Automation Service Setup

A minimal, asynchronous FastAPI daemon configured with systemd to trigger local Android build scripts safely.

Step 1

Set Up Project Directory and Virtual Environment

Create a dedicated directory, set appropriate permissions, and prepare the Python virtual environment.

sudo mkdir -p /opt/builder-api
sudo chown -R james:james /opt/builder-api
cd /opt/builder-api

python3 -m venv venv
source venv/bin/activate
pip install fastapi uvicorn
Step 2

Write the API Server

Save the following Python code to /opt/builder-api/server.py. It executes build jobs asynchronously in the background and returns a tracking job_id immediately.

import subprocess
import uuid
import re
from typing import Optional, Dict
from fastapi import FastAPI, BackgroundTasks, Header, HTTPException, status
from pydantic import BaseModel

app = FastAPI(title="Android Build Trigger API")

# Define your secure pre-shared token
API_KEY = "CHANGE_ME_TO_A_SECURE_RANDOM_STRING"

# In-memory job state tracker
jobs: Dict[str, Dict[str, str]] = {}

# Input models
class CloneRequest(BaseModel):
    repo: str  # Project name or full git URL

class UpdateRequest(BaseModel):
    project: str
    branch_flag: Optional[str] = None  # e.g. "-m", "-n", or explicit branch name

def sanitize_input(val: str) -> bool:
    # Allow alphanumeric, hyphen, underscore, dot, and standard git URL characters
    return bool(re.match(r"^[a-zA-Z0-9_./:-]+$", val))

def run_script_task(job_id: str, script_cmd: list):
    try:
        res = subprocess.run(
            script_cmd,
            capture_output=True,
            text=True,
            timeout=900  # 15 min hard limit
        )
        if res.returncode == 0:
            jobs[job_id] = {
                "status": "success",
                "output": res.stdout[-4000:]  # Keep last 4KB of log
            }
        else:
            jobs[job_id] = {
                "status": "failed",
                "output": res.stderr[-4000:] or res.stdout[-4000:]
            }
    except Exception as e:
        jobs[job_id] = {
            "status": "error",
            "output": str(e)
        }

def authenticate(authorization: Optional[str]):
    if not authorization or authorization != f"Bearer {API_KEY}":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or missing Authorization header"
        )

@app.post("/clone", status_code=status.HTTP_202_ACCEPTED)
def clone_and_build(
    payload: CloneRequest,
    background_tasks: BackgroundTasks,
    authorization: Optional[str] = Header(None)
):
    authenticate(authorization)
    
    if not sanitize_input(payload.repo):
        raise HTTPException(status_code=400, detail="Invalid characters in repo target.")

    job_id = str(uuid.uuid4())
    jobs[job_id] = {"status": "running", "output": ""}
    
    background_tasks.add_task(
        run_script_task,
        job_id,
        ["/opt/julebot/bin/clone_android.sh", payload.repo]
    )
    return {"job_id": job_id, "status": "queued"}

@app.post("/update", status_code=status.HTTP_202_ACCEPTED)
def update_and_build(
    payload: UpdateRequest,
    background_tasks: BackgroundTasks,
    authorization: Optional[str] = Header(None)
):
    authenticate(authorization)

    if not sanitize_input(payload.project):
        raise HTTPException(status_code=400, detail="Invalid characters in project name.")

    update_script = f"/home/james/sprillex/android/{payload.project}/update.sh"
    cmd = [update_script]
    if payload.branch_flag:
        if not sanitize_input(payload.branch_flag):
            raise HTTPException(status_code=400, detail="Invalid characters in branch flag.")
        cmd.append(payload.branch_flag)

    job_id = str(uuid.uuid4())
    jobs[job_id] = {"status": "running", "output": ""}

    background_tasks.add_task(run_script_task, job_id, cmd)
    return {"job_id": job_id, "status": "queued"}

@app.get("/status/{job_id}")
def get_job_status(job_id: str, authorization: Optional[str] = Header(None)):
    authenticate(authorization)
    if job_id not in jobs:
        raise HTTPException(status_code=404, detail="Job not found")
    return {"job_id": job_id, **jobs[job_id]}
Step 3

Create and Enable the Systemd Service

Create the unit configuration file at /etc/systemd/system/android-builder.service:

[Unit]
Description=Android Build Automation FastAPI Service
After=network.target

[Service]
Type=simple
User=julebot
Group=julebot
WorkingDirectory=/opt/builder-api
ExecStart=/opt/builder-api/venv/bin/uvicorn server:app --host 0.0.0.0 --port 8080 --workers 1
Restart=always
RestartSec=5

# Security Hardening
PrivateTmp=true
ProtectSystem=full
NoNewPrivileges=true

[Install]
WantedBy=multi-user.target

Apply permissions, reload the systemd configuration, and enable the service:

# Ensure julebot has permissions over the app directory
sudo chown -R julebot:julebot /opt/builder-api

# Reload and start service
sudo systemctl daemon-reload
sudo systemctl enable --now android-builder.service

# Check execution status
sudo systemctl status android-builder.service
Step 4

Client Interaction Reference

1. Trigger a Clone & Build Job

POST http://<PC_IP>:8080/clone
Headers:
  Authorization: Bearer CHANGE_ME_TO_A_SECURE_RANDOM_STRING
  Content-Type: application/json
Body:
  {"repo": "reviewassistant"}

Response (HTTP 202 Accepted):

{
  "job_id": "c16194b5-6548-43bb-8a60-a29d5b78f447",
  "status": "queued"
}

2. Poll Job Status

GET http://<PC_IP>:8080/status/c16194b5-6548-43bb-8a60-a29d5b78f447
Headers:
  Authorization: Bearer CHANGE_ME_TO_A_SECURE_RANDOM_STRING

Response upon completion (HTTP 200 OK):

{
  "job_id": "c16194b5-6548-43bb-8a60-a29d5b78f447",
  "status": "success",
  "output": "BUILD SUCCESSFUL in 42s\n✅ Build complete: /home/james/sprillex/android/reviewassistant/builds/Reviewassistant-B2026-08-26-23-18.apk"
}
